cmake_minimum_required(VERSION 3.22)
project(VONOX VERSION 0.4.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Standalone DSP engine (no JUCE dependency, fully unit-testable)
add_library(vonox_dsp INTERFACE)
target_include_directories(vonox_dsp INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/src)

option(VONOX_BUILD_TESTS "Build DSP unit tests" ON)
if(VONOX_BUILD_TESTS)
    enable_testing()
    add_executable(vonox_tests tests/dsp_tests.cpp)
    target_link_libraries(vonox_tests PRIVATE vonox_dsp)
    add_test(NAME dsp COMMAND vonox_tests)
    add_executable(vonox_engine_tests tests/engine_tests.cpp)
    target_link_libraries(vonox_engine_tests PRIVATE vonox_dsp)
    add_test(NAME engine COMMAND vonox_engine_tests)
    add_executable(vonox_chain_tests tests/chain_tests.cpp)
    target_link_libraries(vonox_chain_tests PRIVATE vonox_dsp)
    add_test(NAME chain COMMAND vonox_chain_tests)
    add_executable(vonox_fx_tests tests/fx_tests.cpp)
    target_link_libraries(vonox_fx_tests PRIVATE vonox_dsp)
    add_test(NAME fx COMMAND vonox_fx_tests)
    add_executable(vonox_pitch_tests tests/pitch_tests.cpp)
    target_link_libraries(vonox_pitch_tests PRIVATE vonox_dsp)
    add_test(NAME pitch COMMAND vonox_pitch_tests)
    add_executable(vonox_restore_tests tests/restore_tests.cpp)
    target_link_libraries(vonox_restore_tests PRIVATE vonox_dsp)
    add_test(NAME restore COMMAND vonox_restore_tests)
    add_executable(vonox_voicechain_tests tests/voice_chain_tests.cpp)
    target_link_libraries(vonox_voicechain_tests PRIVATE vonox_dsp)
    add_test(NAME voicechain COMMAND vonox_voicechain_tests)
endif()

# VST3 plugin (added in a later phase, requires JUCE via FetchContent)
option(VONOX_BUILD_PLUGIN "Build the VST3 plugin (needs JUCE)" OFF)
if(VONOX_BUILD_PLUGIN)
    add_subdirectory(plugin)
endif()
